Command

A searchable command palette — the cmd-K pattern. Compound API with 9 exports: Command, CommandDialog, CommandInput, CommandList, CommandEmpty, CommandGroup, CommandItem, CommandSeparator, CommandShortcut. Filtering is case-insensitive substring by default (override via filter). Items derive their search value from their text content; pass keywords or an explicit value for reliable matching when children include icons. Keyboard navigation: ↑ / ↓ / Home / End to move, Enter to fire.

Imperative API

import {
  Command, CommandDialog, CommandInput, CommandList,
  CommandEmpty, CommandGroup, CommandItem, CommandSeparator,
  CommandShortcut,
} from '@ui/lib';

function AppShell() {
  const [open, setOpen] = useState(false);

  useEffect(() => {
    const onKey = (e) => {
      if (e.key === 'k' && (e.metaKey || e.ctrlKey)) {
        e.preventDefault();
        setOpen((o) => !o);
      }
    };
    document.addEventListener('keydown', onKey);
    return () => document.removeEventListener('keydown', onKey);
  }, []);

  return (
    <CommandDialog id="cmd-k" open={open} onClose={() => setOpen(false)}>
      <Command id="cmd">
        <CommandInput />
        <CommandList>
          <CommandEmpty>No results.</CommandEmpty>
          <CommandGroup heading="Suggestions">
            <CommandItem onSelect={() => navigate('/calendar')}>
              <Calendar />
              Calendar
            </CommandItem>
          </CommandGroup>
        </CommandList>
      </Command>
    </CommandDialog>
  );
}
